cmake_minimum_required(VERSION 3.28)
project(AOYmirCharacterCapture LANGUAGES C CXX)
if(NOT DEFINED YMIR_SOURCE_DIR)
  message(FATAL_ERROR "Pass the pinned Ymir source path as YMIR_SOURCE_DIR")
endif()
set(Ymir_LIBRARY_ONLY ON CACHE BOOL "" FORCE)
set(Ymir_DEV_BUILD OFF CACHE BOOL "" FORCE)
set(Ymir_ENABLE_DEVLOG OFF CACHE BOOL "" FORCE)
set(Ymir_ENABLE_IPO OFF CACHE BOOL "" FORCE)
set(Ymir_EXTRA_INLINING OFF CACHE BOOL "" FORCE)
set(Ymir_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${YMIR_SOURCE_DIR}" ymir)
include("${CMAKE_CURRENT_SOURCE_DIR}/instrument.cmake")
add_executable(ao-ymir-character-capture main.cpp observer_bridge.hpp vdp2_register_names.inc)
target_compile_features(ao-ymir-character-capture PRIVATE cxx_std_20)
target_include_directories(ao-ymir-character-capture PRIVATE
  "${YMIR_SOURCE_DIR}/capture-cereal/include"
  "${YMIR_SOURCE_DIR}/apps/ymir-sdl3/src")
target_link_libraries(ao-ymir-character-capture PRIVATE ymir::ymir-core bcrypt)
if(MSVC)
  target_compile_options(ao-ymir-character-capture PRIVATE /utf-8 /W3)
endif()
